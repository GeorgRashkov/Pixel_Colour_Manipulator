#<rgb formulas

{ |1|  
  r->[ (r[1]+b[2])%255 ]  
  g->[ (g[0]-r[2])*2 ]  
  b->[ (b[1]+g[2]*0.5) ] 
}

{ |2|  
  r->[ (r>128)*255 ]  
  g->[ (g[1]+g[2]+g[3])/3 ]  
  b->[ (b[0]-r[1]) ] 
}

{ |3|  r->[ 255-r ]  g->[ 255-b ]  b->[255-g ] }


{ |3|  
  r->[ (r[1]-r[2]+r[3]) ]  
  g->[ (g[2]*2-g[1]) ]  
  b->[ (b+r[1])%255 ] 
}

{ |4|  
  r->[ 255-r[1] ]  
  g->[ (b[2]+r[3])*0.5 ]  
  b->[ (g[0]-b[1])*2 ] 
}

{ |5|  
  r->[ r*1.5 + g[1]*0.5 ]  
  g->[ g*1.5 + b[1]*0.5 ]  
  b->[ b*1.5 + r[1]*0.5 ] 
}

{ |6|  
  r->[ (r+g+b)/3 + 100 ]  
  g->[ (g[1]+b[1])*0.7 ]  
  b->[ (r[1]+b)*0.7 ] 
}

{ |7|  
  r->[ (r[1]+r[2])*0.5 ]  
  g->[ 255-(g[0]*0.5) ]  
  b->[ (b[1]+g[2]) ] 
}

{ |8|  
  r->[ (r>100)*200 + g ]  
  g->[ (g>100)*200 + b ]  
  b->[ (b>100)*200 + r ] 
}

{ |9|  
  r->[ r[1]*0.9 ]  
  g->[ g[0]*0.7 ]  
  b->[ b[2]*0.9 ] 
}


{ |10|  r->[ 255-r ]  g->[ 255-b ]  b->[255-g ] }

{ |11|  r->[ r[1] ]  g->[ g[1] ]  b->[ b[1] ] }
